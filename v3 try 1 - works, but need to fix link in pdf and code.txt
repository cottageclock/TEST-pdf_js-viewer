    <div id="photoPane">
      <div id="photoTopBar">
        <button class="btn" id="closeBtn" aria-label="Close photo viewer">×</button>
        <button class="btn" id="fsBtn" aria-label="Toggle full screen">⤢</button>
      </div>

      <button id="leftArrow" aria-label="Previous photo">‹</button>
      <button id="rightArrow" aria-label="Next photo">›</button>

      <div id="photoStage">
        <img id="photoImg" alt="Photo viewer" />
      </div>

      <div id="caption"></div>
      <div id="loading">Loading…</div>
    </div>

  <script>
    // --------- CONFIG ----------
    const PDF_URL = "filename.pdf"; // change if needed
    const FOLDER_API_URL = "https://api.github.com/repos/cottageclock/TEST-pdf_js-viewer/contents/Photos%20folder/Lot%201%20-%20bridge%20scene%20glass%20IMG_4863";
    const PHOTO_FOLDER_WEB_URL = "https://github.com/cottageclock/TEST-pdf_js-viewer/tree/main/Photos%20folder/Lot%201%20-%20bridge%20scene%20glass%20IMG_4863";
    const RAW_BASE = "https://raw.githubusercontent.com/cottageclock/TEST-pdf_js-viewer/main/Photos%20folder/Lot%201%20-%20bridge%20scene%20glass%20IMG_4863/";

    const pdfPane = document.getElementById("pdfPane");
    const photoPane = document.getElementById("photoPane");
    const pdfFrame = document.getElementById("pdfFrame");
    const photoImg = document.getElementById("photoImg");
    const caption = document.getElementById("caption");
    const loading = document.getElementById("loading");

    const leftArrow = document.getElementById("leftArrow");
    const rightArrow = document.getElementById("rightArrow");
    const closeBtn = document.getElementById("closeBtn");
    const fsBtn = document.getElementById("fsBtn");

    let photos = [];
    let currentIndex = 0;

    pdfFrame.src = PDF_URL;

    function showPdf() {
      photoPane.style.display = "none";
      pdfPane.style.display = "block";
    }

    function showPhotos(index = 0) {
      if (!photos.length) return;
      currentIndex = (index + photos.length) % photos.length;
      const file = photos[currentIndex];
      photoImg.src = RAW_BASE + encodeURIComponent(file.name);
      caption.textContent = `${currentIndex + 1} / ${photos.length} — ${file.name}`;
      photoPane.style.display = "block";
      pdfPane.style.display = "none";
    }

    function nextPhoto() {
      if (!photos.length) return;
      showPhotos(currentIndex + 1);
    }

    function prevPhoto() {
      if (!photos.length) return;
      showPhotos(currentIndex - 1);
    }

    async function loadPhotos() {
      loading.style.display = "flex";
      try {
        const res = await fetch(FOLDER_API_URL, {
          headers: { "Accept": "application/vnd.github+json" }
        });
        if (!res.ok) throw new Error("Could not load folder list");

        const data = await res.json();
        photos = data
          .filter(x => x.type === "file" && /\.(jpe?g|png|gif|webp|bmp|avif)$/i.test(x.name))
          .sort((a, b) => a.name.localeCompare(b.name, undefined, { numeric: true }));

        if (!photos.length) throw new Error("No photos found");
      } finally {
        loading.style.display = "none";
      }
    }

    // If the PDF link is clicked and points to the folder, open the photo viewer.
    // Works if your PDF is displayed through a PDF.js viewer or a browser that exposes link clicks.
    window.addEventListener("message", (e) => {
      const data = e.data || {};
      if (data.type === "open-photo-folder" && data.url === PHOTO_FOLDER_WEB_URL) {
        loadPhotos().then(() => showPhotos(0));
      }
    });

    // Fallback: if you want to directly open the gallery from a normal hyperlink on the page.
    // Example: <a href="#" data-photo-folder>Open photos</a>
    document.addEventListener("click", async (e) => {
      const a = e.target.closest("a[data-photo-folder]");
      if (!a) return;
      e.preventDefault();
      await loadPhotos();
      showPhotos(0);
    });

    leftArrow.addEventListener("click", prevPhoto);
    rightArrow.addEventListener("click", nextPhoto);
    closeBtn.addEventListener("click", showPdf);

    fsBtn.addEventListener("click", async () => {
      try {
        if (!document.fullscreenElement) {
          await photoPane.requestFullscreen();
        } else {
          await document.exitFullscreen();
        }
      } catch (_) {}
    });

    document.addEventListener("keydown", (e) => {
      if (photoPane.style.display === "block") {
        if (e.key === "ArrowLeft") prevPhoto();
        if (e.key === "ArrowRight") nextPhoto();
        if (e.key === "Escape") showPdf();
      }
    });

    // Simple tap/click navigation on the image
    photoImg.addEventListener("click", (e) => {
      if (e.clientX < window.innerWidth / 2) prevPhoto();
      else nextPhoto();
    });
  </script>
