<script>
(() => {
  const OWNER = "cottageclock";
  const REPO = "TEST-pdf_js-viewer";
  const DEFAULT_PDF = "filename.pdf";

  const pdfPane = document.getElementById("pdfPane");
  const photoPane = document.getElementById("photoPane");
  const pdfFrame = document.getElementById("pdfFrame");
  const photo = document.getElementById("photo");
  const title = document.getElementById("title");
  const loading = document.getElementById("loading");
  const errorBox = document.getElementById("errorBox");
  const fullOverlay = document.getElementById("fullOverlay");
  const fullImage = document.getElementById("fullImage");

  const prevBtn = document.getElementById("prevBtn");
  const nextBtn = document.getElementById("nextBtn");
  const expandBtn = document.getElementById("expandBtn");
  const closeBtn = document.getElementById("closeBtn");
  const leftTap = document.getElementById("leftTap");
  const rightTap = document.getElementById("rightTap");
  const closeFullBtn = document.getElementById("closeFullBtn");

  let images = [];
  let index = 0;
  let currentFolder = "";

  function showPdf() {
    photoPane.style.display = "none";
    pdfPane.style.display = "block";
    const url = new URL(DEFAULT_PDF, window.location.href).toString();
    pdfFrame.src = url;
  }

  function showGallery() {
    pdfPane.style.display = "none";
    photoPane.style.display = "flex";
  }

  function setLoading(on) {
    loading.style.display = on ? "flex" : "none";
  }

  function showError(msg) {
    errorBox.style.display = "block";
    errorBox.textContent = msg;
  }

  function clearError() {
    errorBox.style.display = "none";
    errorBox.textContent = "";
  }

  function extOk(name) {
    return /\.(jpe?g|png|gif|webp|bmp|avif)$/i.test(name);
  }

  async function loadFolder(folderPath) {
    currentFolder = folderPath;
    setLoading(true);
    clearError();

    const api = `https://api.github.com/repos/${OWNER}/${REPO}/contents/${encodeURI(folderPath)}`;
    const res = await fetch(api, { headers: { "Accept": "application/vnd.github+json" } });

    if (!res.ok) {
      throw new Error("Could not load folder.");
    }

    const data = await res.json();
    const files = Array.isArray(data) ? data : [];
    images = files
      .filter(item => item.type === "file" && extOk(item.name))
      .sort((a, b) => a.name.localeCompare(b.name));

    if (!images.length) {
      throw new Error("No image files found in that folder.");
    }

    index = 0;
    renderImage();
    setLoading(false);
  }

  function renderImage() {
    const item = images[index];
    if (!item) return;

    title.textContent = `${index + 1} / ${images.length} — ${currentFolder.split("/").pop()}`;
    photo.src = item.download_url;
    photo.alt = item.name;
  }

  function next() {
    if (!images.length) return;
    index = (index + 1) % images.length;
    renderImage();
  }

  function prev() {
    if (!images.length) return;
    index = (index - 1 + images.length) % images.length;
    renderImage();
  }

  function openFull() {
    const item = images[index];
    if (!item) return;
    fullImage.src = item.download_url;
    fullOverlay.style.display = "flex";
  }

  function closeFull() {
    fullOverlay.style.display = "none";
    fullImage.src = "";
  }

  prevBtn.onclick = prev;
  nextBtn.onclick = next;
  leftTap.onclick = prev;
  rightTap.onclick = next;
  expandBtn.onclick = openFull;
  closeBtn.onclick = () => {
    const base = new URL(window.location.href);
    base.search = "";
    base.hash = "";
    window.location.href = base.toString();
  };
  closeFullBtn.onclick = closeFull;
  fullOverlay.onclick = (e) => {
    if (e.target === fullOverlay) closeFull();
  };

  photo.onclick = openFull;

  document.addEventListener("keydown", (e) => {
    if (photoPane.style.display !== "flex") return;
    if (fullOverlay.style.display === "flex") {
      if (e.key === "Escape") closeFull();
      return;
    }
    if (e.key === "ArrowLeft") prev();
    if (e.key === "ArrowRight") next();
    if (e.key === "Escape") {
      const base = new URL(window.location.href);
      base.search = "";
      base.hash = "";
      window.location.href = base.toString();
    }
  });

  const params = new URLSearchParams(window.location.search);
  const folder = params.get("folder");

  if (folder) {
    showGallery();
    loadFolder(folder).catch(err => {
      setLoading(false);
      showError(err.message);
    });
  } else {
    showPdf();
  }
})();
