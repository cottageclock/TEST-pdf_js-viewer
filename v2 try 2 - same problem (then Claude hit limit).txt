        // ============ PDF VIEWER ============
        pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

        let pdfDoc = null;
        let pageNum = 1;
        let pageRendering = false;
        let pageNumPending = null;

        const canvas = document.getElementById('pdfCanvas');
        const ctx = canvas.getContext('2d');
        const scale = window.devicePixelRatio || 1;

        async function loadPDF() {
            try {
                const pdfUrl = 'https://raw.githubusercontent.com/cottageclock/TEST-pdf_js-viewer/main/filename.pdf';
                pdfDoc = await pdfjsLib.getDocument(pdfUrl).promise;
                document.getElementById('pageCount').textContent = `of ${pdfDoc.numPages}`;
                renderPage(pageNum);
            } catch (error) {
                const viewer = document.querySelector('.pdf-viewer');
                viewer.innerHTML = `<div class="error">Error loading PDF: ${error.message}</div>`;
            }
        }

        function renderPage(num) {
            if (pageRendering) {
                pageNumPending = num;
                return;
            }
            pageRendering = true;

            pdfDoc.getPage(num).then((page) => {
                const viewport = page.getViewport({ scale: scale * 1.5 });
                canvas.height = viewport.height;
                canvas.width = viewport.width;

                const renderContext = {
                    canvasContext: ctx,
                    viewport: viewport,
                    enableWebGL: true
                };

                page.render(renderContext).promise.then(() => {
                    pageRendering = false;
                    if (pageNumPending !== null) {
                        renderPage(pageNumPending);
                        pageNumPending = null;
                    }
                });
            });

            document.getElementById('pageNum').value = num;
            updateButtonStates();
        }

        function updateButtonStates() {
            document.getElementById('prevPage').disabled = pageNum <= 1;
            document.getElementById('nextPage').disabled = pageNum >= pdfDoc.numPages;
        }

        function nextPage() {
            if (pageNum < pdfDoc.numPages) {
                pageNum++;
                renderPage(pageNum);
            }
        }

        function previousPage() {
            if (pageNum > 1) {
                pageNum--;
                renderPage(pageNum);
            }
        }

        function goToPage() {
            const num = parseInt(document.getElementById('pageNum').value);
            if (num > 0 && num <= pdfDoc.numPages) {
                pageNum = num;
                renderPage(pageNum);
            }
        }

        document.addEventListener('keydown', (e) => {
            const galleryActive = document.getElementById('galleryContainer').classList.contains('active');
            if (galleryActive) {
                if (e.key === 'ArrowRight') nextPhoto();
                if (e.key === 'ArrowLeft') previousPhoto();
                if (e.key === 'Escape') closeGallery();
            } else {
                if (e.key === 'ArrowRight') nextPage();
                if (e.key === 'ArrowLeft') previousPage();
            }
        });

        loadPDF();

        // ============ PHOTO GALLERY ============
        let photos = [];
        let currentPhotoIndex = 0;
        let isFullscreen = false;

        async function openGallery(folderName) {
            try {
                const folderPath = `Photos%20folder/${folderName}`;
                const apiUrl = `https://api.github.com/repos/cottageclock/TEST-pdf_js-viewer/contents/${folderPath}`;
                
                const response = await fetch(apiUrl);
                const contents = await response.json();

                if (!Array.isArray(contents)) {
                    alert('Error: Could not load photos from folder.');
                    return;
                }

                photos = contents
                    .filter(item => {
                        const ext = item.name.split('.').pop().toLowerCase();
                        return ['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext);
                    })
                    .sort((a, b) => a.name.localeCompare(b.name));

                if (photos.length === 0) {
                    alert('No photos found in this folder.');
                    return;
                }

                currentPhotoIndex = 0;
                document.getElementById('galleryTitle').textContent = folderName;
                document.getElementById('photoCount').textContent = photos.length;
                displayPhoto();
                document.getElementById('galleryContainer').classList.add('active');
            } catch (error) {
                alert('Error loading photos: ' + error.message);
            }
        }

        function displayPhoto() {
            const photo = photos[currentPhotoIndex];
            const rawUrl = photo.download_url;
            
            document.getElementById('galleryImage').src = rawUrl;
            document.getElementById('photoNum').textContent = currentPhotoIndex + 1;
            document.getElementById('photoCount').textContent = photos.length;

            document.getElementById('galleryPrev').disabled = currentPhotoIndex === 0;
            document.getElementById('galleryNext').disabled = currentPhotoIndex === photos.length - 1;
        }

        function nextPhoto() {
            if (currentPhotoIndex < photos.length - 1) {
                currentPhotoIndex++;
                displayPhoto();
            }
        }

        function previousPhoto() {
            if (currentPhotoIndex > 0) {
                currentPhotoIndex--;
                displayPhoto();
            }
        }

        function toggleFullscreen() {
            isFullscreen = !isFullscreen;
            const img = document.getElementById('galleryImage');
            img.classList.toggle('fullscreen');
        }

        function closeGallery() {
            isFullscreen = false;
            document.getElementById('galleryImage').classList.remove('fullscreen');
            document.getElementById('galleryContainer').classList.remove('active');
            photos = [];
        }
