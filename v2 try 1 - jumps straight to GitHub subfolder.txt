            <button class="nav-button" id="nextBtn" onclick="nextPhoto()">→</button>
            <button class="fullscreen-toggle" id="fullscreenBtn" onclick="toggleFullscreen()">Expand</button>

    </div>

    <script>
        let photos = [];
        let currentIndex = 0;
        let folderName = '';

        // Get folder name from URL parameter
        const urlParams = new URLSearchParams(window.location.search);
        folderName = urlParams.get('folder') || 'Lot%201%20-%20bridge%20scene%20glass%20IMG_4863';

        // Decode if needed
        if (folderName.includes('%')) {
            folderName = decodeURIComponent(folderName);
        }

        // Initialize gallery
        async function initGallery() {
            const contentDiv = document.getElementById('content');
            
            try {
                // Fetch folder contents from GitHub API
                const folderPath = `Photos folder/${folderName}`;
                const apiUrl = `https://api.github.com/repos/cottageclock/TEST-pdf_js-viewer/contents/${encodeURIComponent(folderPath)}`;
                
                const response = await fetch(apiUrl);
                
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                
                const files = await response.json();
                
                // Filter image files
                const imageExtensions = ['.jpg', '.jpeg', '.png', '.gif', '.webp'];
                photos = files
                    .filter(file => {
                        const ext = file.name.toLowerCase();
                        return imageExtensions.some(imgExt => ext.endsWith(imgExt));
                    })
                    .sort((a, b) => a.name.localeCompare(b.name))
                    .map(file => ({
                        name: file.name,
                        url: file.download_url
                    }));

                if (photos.length === 0) {
                    contentDiv.innerHTML = '<div class="error">No photos found in this folder.</div>';
                    return;
                }

                currentIndex = 0;
                displayPhoto();
                updateControls();

            } catch (error) {
                console.error('Error loading photos:', error);
                contentDiv.innerHTML = `<div class="error">Error loading photos. Please check the folder name or try again later.<br><br>${error.message}</div>`;
            }
        }

        function displayPhoto() {
            if (photos.length === 0) return;

            const photo = photos[currentIndex];
            const contentDiv = document.getElementById('content');
            
            contentDiv.innerHTML = `<img src="${photo.url}" alt="Photo" class="photo-display" title="Click to expand">`;
            document.getElementById('counter').textContent = `${currentIndex + 1} / ${photos.length}`;
            
            // Add click to expand
            contentDiv.querySelector('img').onclick = toggleFullscreen;
        }

        function nextPhoto() {
            if (currentIndex < photos.length - 1) {
                currentIndex++;
                displayPhoto();
                updateControls();
            }
        }

        function previousPhoto() {
            if (currentIndex > 0) {
                currentIndex--;
                displayPhoto();
                updateControls();
            }
        }

        function updateControls() {
            document.getElementById('prevBtn').disabled = currentIndex === 0;
            document.getElementById('nextBtn').disabled = currentIndex === photos.length - 1;
        }

        function toggleFullscreen() {
            if (photos.length === 0) return;
            
            const modal = document.getElementById('modalOverlay');
            const modalImg = document.getElementById('modalImage');
            const photo = photos[currentIndex];
            
            modalImg.src = photo.url;
            modal.classList.add('active');
        }

        function closeFullscreen() {
            const modal = document.getElementById('modalOverlay');
            modal.classList.remove('active');
        }

        function closeGallery() {
            // Go back to the previous page or close the window
            if (document.referrer) {
                window.history.back();
            } else {
                window.close();
            }
        }

        // Keyboard navigation
        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowRight') nextPhoto();
            if (e.key === 'ArrowLeft') previousPhoto();
            if (e.key === 'Escape') closeFullscreen();
        });

        // Initialize on page load
        initGallery();
    </script>
